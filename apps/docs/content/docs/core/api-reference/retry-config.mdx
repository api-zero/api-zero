---
title: RetryConfig
description: Every retry option and what it defaults to.
---

Set on the client or on a single request. **Retries are off until you configure
them** — see [the retries guide](/docs/core/guides/retries) for the policy
behind these options.

<TypeTable
  type={{
    attempts: {
      type: "number",
      required: true,
      description:
        "Total attempts including the first. 1 or less disables retrying.",
    },
    delay: {
      type: "number",
      required: true,
      description: "Base delay in milliseconds, before backoff is applied.",
    },
    maxDelay: {
      type: "number",
      default: "30000",
      description: "Upper bound for any single wait, applied after backoff.",
    },
    backoff: {
      type: '"exponential" | "linear" | fn',
      default: '"exponential"',
      description:
        "exponential gives delay × 2^(attempt-1); linear gives delay × attempt. A function receives (attempt, error) and returns milliseconds.",
    },
    jitter: {
      type: "boolean",
      default: "true",
      description:
        "Multiplies the delay by a random factor between 0.5 and 1, so clients recovering from the same outage do not stampede in lockstep.",
    },
    respectRetryAfter: {
      type: "boolean",
      default: "true",
      description:
        "Obey a Retry-After header on 429 and 503. When present it wins outright and backoff is skipped. Both seconds and HTTP-date forms are parsed.",
    },
    retryMethods: {
      type: "HttpMethod[]",
      default: '["GET", "PUT", "DELETE"]',
      description:
        "Methods eligible for automatic retry. These are the idempotent ones.",
    },
    retryUnsafeMethods: {
      type: "boolean",
      default: "false",
      description:
        "Also allow POST and PATCH. Only for endpoints you know are safe to repeat.",
    },
    retryCondition: {
      type: "(error: ApiError, attempt: number) => boolean",
      description:
        "Replaces the default eligibility test entirely — including the method check. See the warning below.",
    },
    onRetry: {
      type: "(event: RetryEvent) => void",
      description:
        "Called before each wait, with the attempt, the error and the delay about to be taken.",
    },
  }}
/>

<Callout type="warn">
  **`retryCondition` bypasses the method check.** It replaces the whole
  eligibility test, so a condition like `(error) => error.is5xx()` retries a
  `POST` even with `retryUnsafeMethods: false`. If you write one, decide about
  method safety inside it — `error.request?.method` is right there.
</Callout>
